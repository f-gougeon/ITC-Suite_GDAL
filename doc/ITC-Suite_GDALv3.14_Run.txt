

REM 	To run ITC-Suite programs with GDAL version 3.14  (on Ciara and most PCs)

REM   For the projection info is located

set PROJ_LIB=C:\gdal_v3.14\projlib


REM	  Where the GDAL ITC-Suite is located

set PATH=C:\ITC-Suite_GDAL_v3.14\exe;%PATH%
